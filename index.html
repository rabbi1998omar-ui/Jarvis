function speak(text){ if(!("speechSynthesis" in window)){ return; } speechSynthesis.cancel(); const voices = speechSynthesis.getVoices(); /* প্রথমে বাংলা female voice খোঁজা */ let femaleVoice = voices.find(function(voice){ const name = voice.name.toLowerCase(); const lang = voice.lang.toLowerCase(); return ( lang.startsWith("bn") && ( name.includes("female") || name.includes("woman") || name.includes("girl") || name.includes("zira") || name.includes("samantha") || name.includes("google") ) ); }); /* বাংলা যেকোনো voice */ if(!femaleVoice){ femaleVoice = voices.find(function(voice){ return voice.lang && voice.lang.toLowerCase().startsWith("bn"); }); } /* বাংলা voice না পাওয়া গেলে English female-style voice খোঁজা */ if(!femaleVoice){ femaleVoice = voices.find(function(voice){ const name = voice.name.toLowerCase(); return ( name.includes("female") || name.includes("woman") || name.includes("girl") || name.includes("samantha") || name.includes("zira") ); }); } const speech = new SpeechSynthesisUtterance(text); if(femaleVoice){ speech.voice = femaleVoice; speech.lang = femaleVoice.lang; } else{ speech.lang = "bn-BD"; } /* মিষ্টি ও নরম কণ্ঠের জন্য */ speech.rate = 0.88; speech.pitch = 1.32; speech.volume = 1; speechSynthesis.speak( speech ); }
